How it works

main.ts

import { createVideoHost } from "html-video-element-webworker-polyfill/host";

const worker = new Worker(new URL("./render.worker.ts", import.meta.url), { type: "module" });
createVideoHost(worker); // serves <video> elements to the worker

const canvas = document.querySelector("canvas").transferControlToOffscreen();
worker.postMessage({ canvas }, [canvas]);

render.worker.ts

import { polyfillDocument } from "html-video-element-webworker-polyfill/worker";
import { polyfillWebglAPI } from "html-video-element-webworker-polyfill/webgl";
polyfillDocument();
polyfillWebglAPI();

const video = document.createElement("video");
video.src = "https://…/big-buck-bunny.webm";
video.addEventListener("seeked", render);
video.requestVideoFrameCallback(function onFrame() {
  render();
  video.requestVideoFrameCallback(onFrame);
});

function render() {
  gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, video);
  gl.drawArrays(gl.TRIANGLES, 0, 3);
}
